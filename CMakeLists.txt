cmake_minimum_required(VERSION 3.16)
project(MemCore)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 COMPONENTS Widgets REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HUNSPELL REQUIRED hunspell)
add_executable(MemCore main.cpp)
target_include_directories(MemCore PRIVATE ${HUNSPELL_INCLUDE_DIRS})
target_link_libraries(MemCore PRIVATE Qt6::Widgets OpenSSL::SSL OpenSSL::Crypto ${HUNSPELL_LIBRARIES})
